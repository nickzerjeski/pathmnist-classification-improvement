% ------------------------------------------------------------
% 3D box
% ------------------------------------------------------------
\newcommand{\BaseModelLayerBox}[9]{%
\begin{scope}[shift={(#1,#2)}]
    \def\W{#3}
    \def\H{#4}
    \def\D{#5}

    \fill[#6] (0,0) rectangle (\W,\H);

    \fill[#6!85!white]
        (0,\H) -- (\D,\H+\D) -- (\W+\D,\H+\D) -- (\W,\H) -- cycle;

    \fill[#6!75!black!10]
        (\W,0) -- (\W+\D,\D) -- (\W+\D,\H+\D) -- (\W,\H) -- cycle;

    \draw[#7, line width=0.8pt] (0,0) rectangle (\W,\H);
    \draw[#7, line width=0.8pt]
        (0,\H) -- (\D,\H+\D) -- (\W+\D,\H+\D) -- (\W,\H);
    \draw[#7, line width=0.8pt]
        (\W,0) -- (\W+\D,\D) -- (\W+\D,\H+\D) -- (\W,\H);

    \node[font=\scriptsize, align=center] at (\W/2,-0.30) {#8};
    \node[font=\scriptsize, align=center] at (\W/2,-0.68) {#9};
\end{scope}
}

    \fill[#6] (0,0) rectangle (\W,\H);

    \fill[#6!85!white]
        (0,\H) -- (\D,\H+\D) -- (\W+\D,\H+\D) -- (\W,\H) -- cycle;

    \fill[#6!75!black!10]
        (\W,0) -- (\W+\D,\D) -- (\W+\D,\H+\D) -- (\W,\H) -- cycle;

    \draw[#7, line width=0.8pt] (0,0) rectangle (\W,\H);
    \draw[#7, line width=0.8pt]
        (0,\H) -- (\D,\H+\D) -- (\W+\D,\H+\D) -- (\W,\H);
    \draw[#7, line width=0.8pt]
        (\W,0) -- (\W+\D,\D) -- (\W+\D,\H+\D) -- (\W,\H);

    \node[font=\scriptsize, align=center] at (\W/2,-0.32) {#8};
\end{scope}
}
